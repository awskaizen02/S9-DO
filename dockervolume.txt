attach volume to container using -v or --mount

to mount volume into container

docker run --name ngx1 -v vol1:/usr/share/nginx/html:ro -d -P nginx
docker run --name ngx2 -v vol1:/usr/share/nginx/html:ro -d -P nginx

docker run --name ngx6 --mount  type=volume,src=vol1,dst=/usr/share/nginx/html -d -P nginx
